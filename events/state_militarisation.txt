			}
		}
		if = {

				check_variable = {
					var = var:FROM.dmz_state2_var
					value = 0
					compare = not_equals
				}
			}
			var:FROM.dmz_state2_var = {
				set_demilitarized_zone = no
				clr_state_flag = DMZ_recent_militarisation_flag
			}
		}
		if = {
			limit = {
				check_variable = {
					var = var:FROM.dmz_state3_var
					value = 0
					compare = not_equals
				}
			}
			var:FROM.dmz_state3_var = {
				set_demilitarized_zone = no
				clr_state_flag = DMZ_recent_militarisation_flag
			}
		}
		if = {
			limit = {
				check_variable = {
					var = var:FROM.dmz_state4_var
					value = 0
					compare = not_equals
				}
			}
			var:FROM.dmz_state4_var = {
				set_demilitarized_zone = no
				clr_state_flag = DMZ_recent_militarisation_flag
			}
		}
		if = {
			limit = {
				check_variable = {
					var = var:FROM.dmz_state5_var
					value = 0
					compare = not_equals
				}
			}
			var:FROM.dmz_state5_var = {
				set_demilitarized_zone = no
				clr_state_flag = DMZ_recent_militarisation_flag
			}
		}

		hidden_effect = {
			every_owned_state = { # Arraying DMZ owners
				limit = {
					OR = {
						state = FROM.dmz_state1_var
						state = FROM.dmz_state2_var
						state = FROM.dmz_state3_var
						state = FROM.dmz_state4_var
						state = FROM.dmz_state5_var
					}
					check_variable = {
						var = THIS.dmz_owner_tag
						value = 0
						compare = not_equals
					}
				}
				add_to_array = { ROOT.DMZ_event_pass_array = THIS.dmz_owner_tag }
				set_state_flag = SI_dmz_pending_resolve_flag
			}

			every_country = { # Calling DMZ owners
				limit = {
					is_in_array = { ROOT.DMZ_event_pass_array = THIS }
				}
				country_event = { id = dmz.4 days = 1 random_days = 6 }
				remove_from_array = { ROOT.DMZ_event_pass_array = THIS }
			}

			FROM = { # Clearing states vars
				set_variable = { THIS.dmz_state1_var = 0 }
				set_variable = { THIS.dmz_state2_var = 0 }
				set_variable = { THIS.dmz_state3_var = 0 }
				set_variable = { THIS.dmz_state4_var = 0 }
				set_variable = { THIS.dmz_state5_var = 0 }
			}
		}

		ai_chance = {
			base = 10
		}
	}

	option = { # Nope
		name = dmz.3.b

		ai_chance = {
			base = 2
		}
	}
}

				}

			base = 5
		}

			target = FROM

		}
		FROM = {
			country_event = { id = dmz.5 days = 1 random_days = 6 }
		}

		ai_chance = {
			base = 1
		}
	}
}
